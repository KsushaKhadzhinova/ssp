static int SimpleMethod(int a, int b) => a + b;

// Метод с циклом
static int MethodWithLoop(int n)
{
    int sum = 0;
    for (int i = 1; i <= n; i++)

    return sum;
}

// Метод с условием

{
    if (value > 0)
        return "Positive";
    else if (value < 0)
        return "Negative";
    else
        return "Zero";
}

|---|---|
| `static int SimpleMethod(int a, int b) => a + b;` | метод, возвращающий сумму: `int` – тип результата, `(int a, int b)` – параметры, `=> a + b` – «результат равен a + b» (**выражение-тело**), `;` – конец |

### Вспомогательный метод AnalyzeIL

```csharp
// Вспомогательный метод для анализа IL
static void AnalyzeIL(string methodName, Delegate method)
{
    Console.WriteLine($"\n--- Анализ: {methodName} ---");

    var methodInfo = method.Method;
    var methodBody = methodInfo.GetMethodBody();

    if (methodBody == null)
    {
        Console.WriteLine("Тело метода не доступно (Release режим?)");
        Console.WriteLine("Перекомпилируйте в Debug режиме!\n");
        return;
    }

    byte[] ilBytes = methodBody.GetILAsByteArray();

    if (ilBytes == null || ilBytes.Length == 0)
    {
        Console.WriteLine("IL-код не получен");
        return;
    }

    Console.WriteLine($"Размер IL: {ilBytes.Length} байт");
    Console.WriteLine($"Локальные переменные: {methodBody.LocalVariables.Count}");
    Console.WriteLine($"Максимальный размер стека: {methodBody.MaxStackSize}");

    Console.WriteLine("\nIL-инструкции (HEX):");
    for (int i = 0; i < ilBytes.Length; i++)
    {
        Console.Write($"{ilBytes[i]:X2} ");
        if ((i + 1) % 16 == 0) Console.WriteLine();
    }
    Console.WriteLine("\n");
}
```

| Фрагмент | Разбор |
|---|---|
| `static void AnalyzeIL(string methodName, Delegate method)` | метод принимает название для вывода и **делегат** – ссылку на анализируемый метод |
| `var methodInfo = method.Method;` | `.Method` – у делегата свойство, возвращающее описание метода (`MethodInfo`). `var` – тип выберет компилятор |
| `var methodBody = methodInfo.GetMethodBody();` | получить тело метода: IL-байты, локальные переменные, размер стека |
| `if (methodBody == null)` | в Release тело может быть недоступно; сообщить и выйти через `return` |
| `byte[] ilBytes = methodBody.GetILAsByteArray();` | **массив байтов IL**; `byte[]` – массив байтов (числа 0–255) |
| `if (ilBytes == null \|\| ilBytes.Length == 0)` | `\|\|` – «или»: байтов нет или массив пуст |
| `ilBytes.Length` | число элементов (размер IL в байтах) |
| `methodBody.LocalVariables.Count` | сколько локальных переменных |
| `methodBody.MaxStackSize` | максимальная глубина стека вычислений |
| `for (int i = 0; i < ilBytes.Length; i++)` | пройти по всем байтам |
| `Console.Write($"{ilBytes[i]:X2} ");` | вывести байт `ilBytes[i]` в **шестнадцатеричном** виде: `:X2` – заглавные hex-цифры, минимум 2 знака (`0A`, `FF`); `Write` – без перевода строки |
| `if ((i + 1) % 16 == 0) Console.WriteLine();` | `%` – остаток от деления: после каждых 16 байт перенос строки |

### CreateAddMethod: a + b

```csharp
public static Func<int, int, int> CreateAddMethod()
{
    // Создаём динамический метод
    DynamicMethod dynamicMethod = new DynamicMethod(
        "DynamicAdd",
        typeof(int),                    // Возвращаемый тип
        new[] { typeof(int), typeof(int) }, // входные параметры
        typeof(ILGeneratorDemo).Module
    );

    // Получаем ILGenerator
    ILGenerator il = dynamicMethod.GetILGenerator();

    // Генерируем IL-код
    // int Add(int a, int b) => a + b;
    il.Emit(OpCodes.Ldarg_0);      // загрузить первый аргумент
    il.Emit(OpCodes.Ldarg_1);      // загрузить второй аргумент
    il.Emit(OpCodes.Add);          // сложить
    il.Emit(OpCodes.Ret);          // вернуть результат

    // Создаём делегат
    return (Func<int, int, int>)dynamicMethod.CreateDelegate(typeof(Func<int, int, int>));
}
```

| Фрагмент | Разбор |
|---|---|
| `new DynamicMethod("DynamicAdd", typeof(int), new[] { typeof(int), typeof(int) }, typeof(ILGeneratorDemo).Module)` | создать «заготовку» метода: имя (для отладки); тип результата `int`; массив типов параметров `int, int`; модуль (куда «прикрепить» метод) |
| `typeof(int)` | оператор `typeof` возвращает объект `Type` для типа |
| `new[] { ... }` | создание массива с выводом типа элементов |
| `ILGenerator il = dynamicMethod.GetILGenerator();` | получить «записывающее устройство» IL |
| `il.Emit(OpCodes.Ldarg_0);` | `Emit` – записать команду; `OpCodes.Ldarg_0` – загрузить аргумент 0 в стек |
| `il.Emit(OpCodes.Ldarg_1);` | аргумент 1 в стек |
| `il.Emit(OpCodes.Add);` | сложить два верхних значения стека |
| `il.Emit(OpCodes.Ret);` | вернуть значение |
| `(Func<int, int, int>)dynamicMethod.CreateDelegate(typeof(Func<int, int, int>))` | создать делегат нужного типа; `(Func<...>)` – явное **приведение типа** (результат `CreateDelegate` имеет тип `Delegate`) |

### CreateFactorialMethod: цикл

```csharp
public static Func<int, int> CreateFactorialMethod()
{
    DynamicMethod dynamicMethod = new DynamicMethod(
        "DynamicFactorial",
        typeof(int),
        new[] { typeof(int) },
        typeof(ILGeneratorDemo).Module
    );

    ILGenerator il = dynamicMethod.GetILGenerator();

    // Метки для переходов
    Label startLoop = il.DefineLabel();
    Label endLoop = il.DefineLabel();

    // Локальные переменные
    il.DeclareLocal(typeof(int)); // result
    il.DeclareLocal(typeof(int)); // i

    // result = 1;
    il.Emit(OpCodes.Ldc_I4_1);
    il.Emit(OpCodes.Stloc_0);

    // i = 1;
    il.Emit(OpCodes.Ldc_I4_1);
    il.Emit(OpCodes.Stloc_1);

    // startLoop:
    il.MarkLabel(startLoop);

    // if (i > arg0) goto endLoop;
    il.Emit(OpCodes.Ldloc_1);
    il.Emit(OpCodes.Ldarg_0);
    il.Emit(OpCodes.Bgt, endLoop);

    // result *= i;
    il.Emit(OpCodes.Ldloc_0);
    il.Emit(OpCodes.Ldloc_1);
    il.Emit(OpCodes.Mul);
    il.Emit(OpCodes.Stloc_0);

    // i++;
    il.Emit(OpCodes.Ldloc_1);
    il.Emit(OpCodes.Ldc_I4_1);
    il.Emit(OpCodes.Add);
    il.Emit(OpCodes.Stloc_1);

    // goto startLoop;
    il.Emit(OpCodes.Br, startLoop);

    // endLoop:
    il.MarkLabel(endLoop);

    // return result;
    il.Emit(OpCodes.Ldloc_0);
    il.Emit(OpCodes.Ret);

    return (Func<int, int>)dynamicMethod.CreateDelegate(typeof(Func<int, int>));
}
```

| Фрагмент | Разбор |
|---|---|
| `Label startLoop = il.DefineLabel();` | создать **метку** – адрес в коде, на который можно перейти |
| `il.DeclareLocal(typeof(int));` | объявить локальную переменную типа `int`; первая получает номер 0 (`result`), вторая 1 (`i`) |
| `OpCodes.Ldc_I4_1` | положить в стек константу 1 (`ldc.i4.1`) |
| `OpCodes.Stloc_0` | сохранить из стека в локальную переменную 0 |
| `il.MarkLabel(startLoop);` | «поставить» метку в текущую позицию кода |
| `OpCodes.Ldloc_1`, `OpCodes.Ldarg_0`, `OpCodes.Bgt, endLoop` | загрузить `i`, загрузить `n`, **если `i > n` перейти на метку `endLoop`** (`Bgt` – branch if greater than) |
| `OpCodes.Mul` | умножить: `result * i` |
| `OpCodes.Br, startLoop` | безусловный переход в начало цикла |
| `il.MarkLabel(endLoop);` затем `Ldloc_0` и `Ret` | выход из цикла, вернуть `result` |

Эквивалент на C#: `int result = 1; int i = 1; while (i <= n) { result *= i; i++; } return result;`.

### CreateGradeMethod: ветвление

```csharp
public static Func<int, string> CreateGradeMethod()
{
    DynamicMethod dynamicMethod = new DynamicMethod(
        "DynamicGrade",
        typeof(string),
        new[] { typeof(int) },
        typeof(ILGeneratorDemo).Module
    );

    ILGenerator il = dynamicMethod.GetILGenerator();

    Label positive = il.DefineLabel();
    Label negative = il.DefineLabel();

    // if (value > 0) goto positive;
    il.Emit(OpCodes.Ldarg_0);
    il.Emit(OpCodes.Ldc_I4_0);
    il.Emit(OpCodes.Bgt, positive);

    // if (value < 0) goto negative;
    il.Emit(OpCodes.Ldarg_0);
    il.Emit(OpCodes.Ldc_I4_0);
    il.Emit(OpCodes.Blt, negative);

    // return "Zero";
    il.Emit(OpCodes.Ldstr, "Zero");
    il.Emit(OpCodes.Ret);

    // positive:
    il.MarkLabel(positive);
    il.Emit(OpCodes.Ldstr, "Positive");
    il.Emit(OpCodes.Ret);

    // negative:
    il.MarkLabel(negative);
    il.Emit(OpCodes.Ldstr, "Negative");
    il.Emit(OpCodes.Ret);

    return (Func<int, string>)dynamicMethod.CreateDelegate(typeof(Func<int, string>));
}
```

| Фрагмент | Разбор |
|---|---|
| `Ldarg_0`, `Ldc_I4_0`, `Bgt, positive` | если `value > 0`, перейти на метку `positive` |
| `Blt, negative` | если `value < 0`, перейти на `negative` |
| `OpCodes.Ldstr, "Zero"` | положить в стек строку «Zero» (`ldstr`) и вернуть (`Ret`) – случай «ноль» |
| `positive: / negative:` | возвращают «Positive» / «Negative» |

### Задание 2.1: рекурсивный факториал

```csharp
// Задание 2.1: n! рекурсивно
// int Factorial(int n) { if (n <= 1) return 1; return n * Factorial(n - 1); }
public static Func<int, int> CreateRecursiveFactorialMethod()
{
    DynamicMethod dynamicMethod = new DynamicMethod(
        "RecursiveFactorial",
        typeof(int),
        new[] { typeof(int) },
        typeof(ILGeneratorDemo).Module
    );

    ILGenerator il = dynamicMethod.GetILGenerator();

    Label recurse = il.DefineLabel();

    // if (n > 1) goto recurse;
    il.Emit(OpCodes.Ldarg_0);
    il.Emit(OpCodes.Ldc_I4_1);
    il.Emit(OpCodes.Bgt, recurse);

    // return 1;
    il.Emit(OpCodes.Ldc_I4_1);
    il.Emit(OpCodes.Ret);

    // recurse: return n * Factorial(n - 1);
    il.MarkLabel(recurse);
    il.Emit(OpCodes.Ldarg_0);
    il.Emit(OpCodes.Ldarg_0);
    il.Emit(OpCodes.Ldc_I4_1);
    il.Emit(OpCodes.Sub);
    il.Emit(OpCodes.Call, dynamicMethod);   // рекурсивный вызов самого метода
    il.Emit(OpCodes.Mul);
    il.Emit(OpCodes.Ret);

    return (Func<int, int>)dynamicMethod.CreateDelegate(typeof(Func<int, int>));
}
```

| Фрагмент | Разбор |
|---|---|
| `Bgt, recurse` | если `n > 1`, перейти к рекурсивной ветке |
| `Ldc_I4_1`, `Ret` | иначе (n ≤ 1) вернуть 1 |
| `Ldarg_0` (первый) | положить `n` – это **левый множитель** для будущего умножения |
| `Ldarg_0`, `Ldc_I4_1`, `Sub` | вычислить `n - 1` |
| `il.Emit(OpCodes.Call, dynamicMethod)` | **вызвать сам этот метод** с аргументом `n - 1` (рекурсия); результат останется в стеке |
| `OpCodes.Mul`, `Ret` | перемножить `n` и результат рекурсии, вернуть |

### Задание 2.2: максимум из трёх

```csharp
// Задание 2.2: максимум из трёх чисел
// int Max(int a, int b, int c) { int max = a; if (b > max) max = b; if (c > max) max = c; return max; }
public static Func<int, int, int, int> CreateMaxOfThreeMethod()
{
    DynamicMethod dynamicMethod = new DynamicMethod(
        "DynamicMax",
        typeof(int),
        new[] { typeof(int), typeof(int), typeof(int) },
        typeof(ILGeneratorDemo).Module
    );

    ILGenerator il = dynamicMethod.GetILGenerator();

    Label skipB = il.DefineLabel();
    Label skipC = il.DefineLabel();

    il.DeclareLocal(typeof(int)); // max

    // max = a;
    il.Emit(OpCodes.Ldarg_0);
    il.Emit(OpCodes.Stloc_0);

    // if (b <= max) goto skipB; max = b;
    il.Emit(OpCodes.Ldarg_1);
    il.Emit(OpCodes.Ldloc_0);
    il.Emit(OpCodes.Ble, skipB);
    il.Emit(OpCodes.Ldarg_1);
    il.Emit(OpCodes.Stloc_0);
    il.MarkLabel(skipB);

    // if (c <= max) goto skipC; max = c;
    il.Emit(OpCodes.Ldarg_2);
    il.Emit(OpCodes.Ldloc_0);
    il.Emit(OpCodes.Ble, skipC);
    il.Emit(OpCodes.Ldarg_2);
    il.Emit(OpCodes.Stloc_0);
    il.MarkLabel(skipC);

    // return max;
    il.Emit(OpCodes.Ldloc_0);
    il.Emit(OpCodes.Ret);

    return (Func<int, int, int, int>)dynamicMethod.CreateDelegate(typeof(Func<int, int, int, int>));
}
```

| Фрагмент | Разбор |
|---|---|
| `il.DeclareLocal(typeof(int)); // max` | локальная переменная `max` |
| `Ldarg_0`, `Stloc_0` | `max = a` |
| `Ldarg_1`, `Ldloc_0`, `Ble, skipB` | если `b <= max`, пропустить присваивание (`Ble` – branch if less or equal) |
| `Ldarg_1`, `Stloc_0` | иначе `max = b` |
| то же для третьего параметра `Ldarg_2` | `max = c`, если `c > max` |
| `Ldloc_0`, `Ret` | вернуть `max` |

```csharp
public static class JitResearch

    public static void Run()

            for (int i = 0; i < 10000; i++)

            sw.Stop();

            Console.WriteLine($"  Время: {sw.ElapsedMilliseconds} мс");
            Console.WriteLine($"  Тики: {sw.ElapsedTicks:N0}");
            Console.WriteLine();
        }

        JitVisualization.InspectMethod(typeof(Calculator).GetMethod("Add")!);
        JitVisualization.CompareJitModes();

        Console.WriteLine("\n=== Профилирование производительности ===\n");

        const int iterations = 10_000_000;
        var timer = new Stopwatch();

        // Direct call
        timer.Start();
        int sum1 = 0;
        for (int i = 0; i < iterations; i++)
            sum1 += i;
        timer.Stop();
        Console.WriteLine($"Прямой вызов: {timer.ElapsedMilliseconds} мс");

        // Through delegate
        Func<int, int, int> add = (x, y) => x + y;
        timer.Restart();
        int sum2 = 0;
        for (int i = 0; i < iterations; i++)
            sum2 = add(sum2, i);
        timer.Stop();
        Console.WriteLine($"Через делегат: {timer.ElapsedMilliseconds} мс");

        // Through virtual method
        var calc = new Calculator();
        timer.Restart();
        int sum3 = 0;
        for (int i = 0; i < iterations; i++)
            sum3 = calc.Add(sum3, i);
        timer.Stop();
        Console.WriteLine($"Через виртуальный метод: {timer.ElapsedMilliseconds} мс");

        ArraySumTask();
    }
```

|---|---|

### Задание 4.1: ArraySumTask и SumArray

```csharp
// Задание 4.1: сложение элементов массива из 10000 целых чисел
// Первый вызов включает JIT-компиляцию метода, последующие выполняют уже скомпилированный код
static void ArraySumTask()
{
    Console.WriteLine("\n=== Задание 4.1: сумма элементов массива (10000 элементов) ===\n");

    int[] array = new int[10000];
    for (int i = 0; i < array.Length; i++)
        array[i] = i;

    // Первый вызов: JIT-компиляция + выполнение
    long start = Stopwatch.GetTimestamp();
    long result = SumArray(array);
    long firstCall = Stopwatch.GetTimestamp() - start;
    Console.WriteLine($"Сумма = {result}");
    Console.WriteLine($"Первый вызов (с JIT-компиляцией): {firstCall * 1_000_000.0 / Stopwatch.Frequency:F1} мкс");

    // Повторные вызовы: метод уже скомпилирован
    const int calls = 1000;
    start = Stopwatch.GetTimestamp();
    for (int i = 0; i < calls; i++)
        SumArray(array);
    long repeated = Stopwatch.GetTimestamp() - start;
    Console.WriteLine($"Повторный вызов (без JIT-компиляции): {repeated * 1_000_000.0 / Stopwatch.Frequency / calls:F1} мкс");
}

[MethodImpl(MethodImplOptions.NoInlining)]
static long SumArray(int[] array)
{
    long sum = 0;
    for (int i = 0; i < array.Length; i++)
        sum += array[i];
    return sum;
}
```

| Фрагмент | Разбор |
|---|---|
| `int[] array = new int[10000];` | массив из 10 000 целых |
| `Stopwatch.GetTimestamp()` | текущее значение высокоточного счётчика |
| первый вызов `SumArray(array)` | включает **JIT-компиляцию** метода |
| `calls = 1000` и цикл | повторные вызовы – метод уже скомпилирован |
| `firstCall * 1_000_000.0 / Stopwatch.Frequency` | перевод «тиков» в микросекунды: умножить на 10⁶ и поделить на частоту счётчика |
| `[MethodImpl(MethodImplOptions.NoInlining)]` | **атрибут**: запретить JIT встраивать метод, чтобы замер был честным |
| `long sum = 0;` | `long` – 64-битное целое, чтобы сумма не переполнилась |
