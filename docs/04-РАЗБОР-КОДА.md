```csharp
public class EventLeakExample
{
    public event EventHandler? BigEvent;
    private List<EventHandler> _handlers = new List<EventHandler>();

    public void SubscribeLeak()
    {
        EventHandler handler = (sender, e) =>
        {
            var data = new byte[1024 * 100];
        };
        BigEvent += handler;
        _handlers.Add(handler);
    }

    public void UnsubscribeAll()
    {
        foreach (var handler in _handlers)
            BigEvent -= handler;
        _handlers.Clear();
    }
}

```

| Фрагмент | Разбор |
|---|---|
| `EventHandler handler = (sender, e) => { var data = new byte[1024 * 100]; };` | обработчик сохранён в **переменной**, чтобы потом можно было отписаться |
| `BigEvent += handler; _handlers.Add(handler);` | подписка и запоминание |
| `foreach (var handler in _handlers) BigEvent -= handler;` | отписка от всех |
| `_handlers.Clear();` | очистить список |

| Фрагмент | Разбор |
|---|---|
| `private static List<IDisposable> _cache` | статический кэш: корень GC |
| `private static event EventHandler? _globalEvent;` | статическое событие: корень GC |
| `private byte[] _data = new byte[1024 * 1024];` | 1 МБ на объект |
| `_globalEvent += OnEvent;` (**утечка 1**) | событие хранит ссылку на объект, пока не отпишемся |

| **утечка 3** | массив 1 МБ достижим через подписку, поэтому не собирается |

```csharp
public class FixedCode : IDisposable
{
    private static List<IDisposable> _cache = new List<IDisposable>();
    private static event EventHandler? _globalEvent;
    private byte[]? _data = new byte[1024 * 1024];
    private MemoryStream _stream = new MemoryStream(1024);

    public FixedCode()
    {
        _globalEvent += OnEvent;
        _cache.Add(_stream);
    }

    private void OnEvent(object? sender, EventArgs e)
    {
        var temp = new byte[1024 * 100];
    }

    public void Dispose()
    {
        _globalEvent -= OnEvent;   // отписка от события
        _cache.Remove(_stream);    // удаление из статического кэша
        _stream.Dispose();         // освобождение потока
        _data = null;
    }

    public static int CacheCount => _cache.Count;
}
```

`FixedCode` реализует `IDisposable`: в `Dispose()` выполняются отписка (`-=`), удаление из кэша (`Remove`), освобождение потока (`Dispose`) и обнуление ссылки на массив.

`LeakSearch.Run()` создаёт 100 объектов `BuggyCode` (≈ 100 МБ остаётся), `LeakSearch.RunFixed()` – 100 объектов `FixedCode` в `using` (кэш пуст).

public static class CallTree
{
    private static List<string> _logs = new List<string>();

### GenerateData: создание данных

```csharp
private static void GenerateData(int count)
{
    var sw = Stopwatch.StartNew();

    var chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
    for (int i = 0; i < count; i++)
    {
        var sb = new StringBuilder();
        sb.Append($"USER{_rnd.Next(1, 100)}: ");
        sb.Append($"{DateTime.Now.AddMinutes(-_rnd.Next(0, 10000)):yyyy-MM-dd HH:mm:ss.fff} | ");
        sb.Append(_rnd.Next(0, 4) switch
        {
            0 => "INFO | ",
            1 => "WARNING | ",
            2 => "ERROR | ",
            _ => "DEBUG | "
        });
        for (int j = 0; j < 50; j++)
            sb.Append(chars[_rnd.Next(chars.Length)]);
        sb.Append($" | Code: {_rnd.Next(100, 999)}");
        _logs.Add(sb.ToString());
    }
    for (int i = 0; i < 100; i++)
        _logs.Add($"ANOMALY: {Guid.NewGuid()} | CRITICAL | Ошибка");

    sw.Stop();
    AddCall("GenerateData", sw.Elapsed.TotalMilliseconds);
}
```

| Фрагмент | Разбор |
|---|---|
| `new StringBuilder()` | буфер для сборки строки |
| `sb.Append($"USER{_rnd.Next(1, 100)}: ")` | `Next(1, 100)` – случайное число от 1 до 99 |
| `DateTime.Now.AddMinutes(-_rnd.Next(0, 10000))` | момент времени «сейчас минус случайное число минут» |
| `{... :yyyy-MM-dd HH:mm:ss.fff}` | **формат даты** в интерполяции: год-месяц-день часы:минуты:секунды.миллисекунды |
| `_rnd.Next(0, 4) switch { 0 => "INFO | ", ... }` | **switch-выражение**: по значению выбирает строку уровня |
| `for (int j = 0; j < 50; j++) sb.Append(chars[...])` | 50 случайных символов |
| `$" | Code: {...}"` | код 100–998 |
| `_logs.Add($"ANOMALY: {Guid.NewGuid()} | CRITICAL | Ошибка");` | ещё 100 «аномальных» записей |

Формат строки журнала: `USER5: 2026-10-08 12:00:00.123 | ERROR | <50 символов> | Code: 512`.

### ProcessLogs: фильтрация и анализ

```csharp
private static List<string> ProcessLogs()
{
    var sw = Stopwatch.StartNew();
    var anomalies = new List<string>();

    // Уровень 1: Фильтрация
    var filterSw = Stopwatch.StartNew();
    var filtered = new List<string>();
    foreach (var log in _logs)
    {
        if (log.IndexOf("ERROR", StringComparison.OrdinalIgnoreCase) >= 0 ||
            log.IndexOf("WARNING", StringComparison.OrdinalIgnoreCase) >= 0 ||
            log.IndexOf("CRITICAL", StringComparison.OrdinalIgnoreCase) >= 0)
        {
            filtered.Add(log);
        }
    }
    filterSw.Stop();
    AddCall("FilterLogs (IndexOf)", filterSw.Elapsed.TotalMilliseconds);

    // Уровень 2: Анализ аномалий (САМЫЙ ГОРЯЧИЙ)
    var anomalySw = Stopwatch.StartNew();
    foreach (var log in filtered)
    {
        if (IsAnomaly(log))
            anomalies.Add(log);
    }
    anomalySw.Stop();
    AddCall("IsAnomaly (total with children)", anomalySw.Elapsed.TotalMilliseconds);

    sw.Stop();
    AddCall("ProcessLogs", sw.Elapsed.TotalMilliseconds);
    return anomalies;
}
```

| Фрагмент | Разбор |
|---|---|
| `log.IndexOf("ERROR", StringComparison.OrdinalIgnoreCase) >= 0` | поиск подстроки без учёта регистра; `>= 0` – найдена |
| `\|\|` | «или»: оставляем ERROR, WARNING, CRITICAL |
| `AddCall("FilterLogs (IndexOf)", ...)` | записать время шага в словарь |
| цикл `foreach (var log in filtered) if (IsAnomaly(log)) ...` | анализ каждой отфильтрованной записи |

### IsAnomaly: пять шагов

```csharp
private static bool IsAnomaly(string log)
{
    var methodSw = Stopwatch.StartNew();
    bool result = false;

    // Шаг 1: Проверка длины (быстро)
    if (log.Length < 100)
    {
        methodSw.Stop();
        AddCall("IsAnomaly.CheckLength", methodSw.Elapsed.TotalMilliseconds);
        return false;
    }

    // Шаг 2: String.Split (средняя тяжесть)
    var splitSw = Stopwatch.StartNew();
    var parts = log.Split('|');
    splitSw.Stop();
    AddCall("IsAnomaly.String.Split", splitSw.Elapsed.TotalMilliseconds);

    if (parts.Length < 3)
    {
        methodSw.Stop();
        AddCall("IsAnomaly", methodSw.Elapsed.TotalMilliseconds);
        return false;
    }

    // Шаг 3: разбор даты (тяжелый). Дата находится в первой части строки после "USERnn:"
    string dateText = parts[0].Substring(parts[0].IndexOf(':') + 1).Trim();
    var dateSw = Stopwatch.StartNew();
    DateTime logDate = default;
    bool dateOk;
    if (_parseExact)
    {
        // Задание 2: DateTime.ParseExact с фиксированным форматом
        try
        {
            logDate = DateTime.ParseExact(dateText, "yyyy-MM-dd HH:mm:ss.fff", CultureInfo.InvariantCulture);
            dateOk = true;
        }
        catch (FormatException)
        {
            dateOk = false;
        }
    }
    else
    {
        dateOk = DateTime.TryParse(dateText, out logDate);
    }
    dateSw.Stop();
    AddCall(_parseExact ? "IsAnomaly.DateTime.ParseExact" : "IsAnomaly.DateTime.TryParse", dateSw.Elapsed.TotalMilliseconds);

    if (!dateOk)
    {
        methodSw.Stop();
        AddCall("IsAnomaly", methodSw.Elapsed.TotalMilliseconds);
        return false;
    }

    // Шаг 4: Regex (САМЫЙ ТЯЖЕЛЫЙ!)
    var regexSw = Stopwatch.StartNew();
    var codeMatch = Regex.Match(parts.Length > 3 ? parts[3] : "", @"Code:\s*(\d+)");
    var code = codeMatch.Success && int.TryParse(codeMatch.Groups[1].Value, out int c) ? c : 0;
    regexSw.Stop();
    AddCall("IsAnomaly.Regex.Match", regexSw.Elapsed.TotalMilliseconds);

    // Шаг 5: Проверки (легкие)
    var checkSw = Stopwatch.StartNew();
    var hasAnomaly = log.Contains("ANOMALY", StringComparison.OrdinalIgnoreCase);
    var isRecent = logDate > DateTime.Now.AddDays(-1);
    result = (hasAnomaly && isRecent) || (code > 500 && isRecent) || (hasAnomaly && code > 500);
    checkSw.Stop();
    AddCall("IsAnomaly.Checks", checkSw.Elapsed.TotalMilliseconds);

    methodSw.Stop();
    AddCall("IsAnomaly (own time)", methodSw.Elapsed.TotalMilliseconds);
    return result;
}
```

| Шаг | Разбор |
|---|---|
| Шаг 1 `log.Length < 100` | слишком короткие строки (например, аномалии без поля данных) отбрасываются |
| Шаг 2 `log.Split('\|')` | разрезать строку по символу `\|` на части (`parts`) |
| Шаг 3 | **дата**: `parts[0].Substring(parts[0].IndexOf(':') + 1).Trim()` – взять часть строки после первого двоеточия (после «USERnn:») и убрать пробелы. Затем либо `DateTime.TryParse(dateText, out logDate)` – разбор «как получится», возвращает `true/false`; либо `DateTime.ParseExact(dateText, "yyyy-MM-dd HH:mm:ss.fff", CultureInfo.InvariantCulture)` – строго по формату, при несоответствии выбрасывает `FormatException` (перехватывается) |
| Шаг 4 | `Regex.Match(parts[3], @"Code:\s*(\d+)")` – найти «Code:» и число; `\s*` – любые пробелы, `(\d+)` – группа из цифр. Дорогая операция |
| Шаг 5 | проверки: содержит ли «ANOMALY», недавняя ли дата, код больше 500 |
| `out logDate` | **выходной параметр**: метод записывает результат в переменную |

### Вывод дерева

```csharp
private static void AddCall(string methodName, double timeMs)
{
    if (!_callCounts.ContainsKey(methodName))
    {
        _callCounts[methodName] = 0;
        _callTimes[methodName] = 0;
    }
    _callCounts[methodName]++;
    _callTimes[methodName] += timeMs;
}

private static double Time(string name) => _callTimes.TryGetValue(name, out var t) ? t : 0;

    // Корень
    PrintNode("Main", totalTime, totalTime);

    // Первый уровень
    PrintNode("  |-- GenerateData", Time("GenerateData"), totalTime);
    PrintNode("  `-- ProcessLogs", Time("ProcessLogs"), totalTime);

    // Второй уровень (дети ProcessLogs)
    PrintNode("      |-- FilterLogs (IndexOf)", Time("FilterLogs (IndexOf)"), Time("ProcessLogs"));
    PrintNode("      `-- IsAnomaly (total with children)", Time("IsAnomaly (total with children)"), Time("ProcessLogs"));

    // Третий уровень (дети IsAnomaly)
    double isAnomalyTotal = Time("IsAnomaly (total with children)");
    PrintNode("          |-- IsAnomaly.String.Split", Time("IsAnomaly.String.Split"), isAnomalyTotal);
    PrintNode("          |-- " + dateKey, Time(dateKey), isAnomalyTotal);
    PrintNode("          |-- IsAnomaly.Regex.Match", Time("IsAnomaly.Regex.Match"), isAnomalyTotal);
    PrintNode("          |-- IsAnomaly.Checks", Time("IsAnomaly.Checks"), isAnomalyTotal);
    PrintNode("          `-- IsAnomaly (own time)", Time("IsAnomaly (own time)"), isAnomalyTotal);

    Console.WriteLine("\n=================== Горячий путь (Hot Path) ===================");
    Console.WriteLine("Main -> ProcessLogs -> IsAnomaly -> " + (Time("IsAnomaly.Regex.Match") > Time(dateKey) ? "Regex.Match" : dateKey));

    Console.WriteLine($"\nСтатистика вызовов:");
    foreach (var kvp in _callCounts)
        Console.WriteLine($"   {kvp.Key}: {kvp.Value} вызовов, {_callTimes[kvp.Key]:F1} мс");
}
```

| Фрагмент | Разбор |
|---|---|
| `AddCall(string methodName, double timeMs)` | если шага ещё нет в словаре – создать запись с нулями; затем прибавить 1 к числу вызовов и `timeMs` к общему времени |
| `Time(string name) => _callTimes.TryGetValue(name, out var t) ? t : 0;` | безопасное чтение: если нет ключа – 0 (в методичке здесь было падение) |
| `PrintNode(name, time, parentTime)` | печатает строку: имя, время, долю `time / parentTime * 100` и столбик из `#` длиной `процент / 2` |
| `Горячий путь` | сравниваются времена `Regex.Match` и разбора даты, печатается более дорогой |

    var sw = Stopwatch.StartNew();

private static string BuildTextWithBuilder(int count)
{
    var sw = Stopwatch.StartNew();
    var sb = new StringBuilder();
    for (int i = 0; i < count; i++)
        sb.Append("слово ");
    Add("BuildTextWithBuilder", sw);
    return sb.ToString();
}

private static int CountWords(string text)
{
    var sw = Stopwatch.StartNew();
    int count = text.Split(' ', StringSplitOptions.RemoveEmptyEntries).Length;
    Add("CountWords", sw);
